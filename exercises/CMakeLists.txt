cmake_minimum_required(VERSION 3.18)

# Gadi gpuvolta uses V100 GPUs. Override for another GPU architecture.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT DEFINED ENV{CUDAARCHS})
    set(CMAKE_CUDA_ARCHITECTURES 70 CACHE STRING "CUDA architectures to build for")
endif()
project(cuda_exercises LANGUAGES C CXX CUDA)
find_package(CUDAToolkit REQUIRED)
option(EXERCISES_BUILD_MKL "Build the CPU Intel MKL GEMM example" ON)
set(EXERCISES_EXTRA_SOURCES "" CACHE STRING "Semicolon-separated generated notebook sources, relative to the repository root")

if(EXERCISES_BUILD_MKL)
    find_path(MKL_INCLUDE_DIR mkl.h HINTS ENV MKLROOT PATH_SUFFIXES include)
    foreach(component IN ITEMS intel_lp64 sequential core)
        find_library(MKL_${component}_LIBRARY NAMES mkl_${component}
            HINTS ENV MKLROOT PATH_SUFFIXES lib/intel64 lib)
        if(NOT MKL_${component}_LIBRARY)
            message(FATAL_ERROR "MKL library mkl_${component} not found. Load intel-mkl or set EXERCISES_BUILD_MKL=OFF.")
        endif()
    endforeach()
    if(NOT MKL_INCLUDE_DIR)
        message(FATAL_ERROR "mkl.h not found. Load intel-mkl or set EXERCISES_BUILD_MKL=OFF.")
    endif()
    find_package(Threads REQUIRED)
endif()

# Only numbered exercise folders are scanned; build/compiler probes are excluded.
file(GLOB exercise_dirs CONFIGURE_DEPENDS LIST_DIRECTORIES TRUE
    "${CMAKE_CURRENT_SOURCE_DIR}/[0-9][0-9]-*")
set(sources)
foreach(exercise_dir IN LISTS exercise_dirs)
    file(GLOB_RECURSE exercise_sources CONFIGURE_DEPENDS
        "${exercise_dir}/*.cu" "${exercise_dir}/*.c" "${exercise_dir}/*.cpp")
    list(APPEND sources ${exercise_sources})
endforeach()
foreach(extra IN LISTS EXERCISES_EXTRA_SOURCES)
    get_filename_component(extra "${extra}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
    if(NOT EXISTS "${extra}")
        message(FATAL_ERROR "Generated notebook source does not exist: ${extra}")
    endif()
    list(APPEND sources "${extra}")
endforeach()
list(REMOVE_DUPLICATES sources)

foreach(source IN LISTS sources)
    get_filename_component(stem "${source}" NAME_WE)
    get_filename_component(extension "${source}" EXT)
    get_filename_component(source_dir "${source}" DIRECTORY)
    file(RELATIVE_PATH folder "${CMAKE_CURRENT_SOURCE_DIR}" "${source_dir}")
    if(stem STREQUAL "perf-gemm-intelmkl" AND NOT EXERCISES_BUILD_MKL)
        continue()
    endif()
    if(folder MATCHES "^[0-9][0-9]-")
        get_filename_component(last_folder "${folder}" NAME)
        if(folder MATCHES "/" AND last_folder MATCHES "^[0-9][0-9]-${stem}$")
            set(target "ex-${folder}")
        else()
            set(target "ex-${folder}-${stem}")
        endif()
    else()
        set(target "ex-extra-${stem}")
    endif()
    string(REPLACE "/" "-" target "${target}")
    if(TARGET ${target})
        message(FATAL_ERROR "Duplicate executable target: ${target}")
    endif()
    add_executable(${target} "${source}")
    set_target_properties(${target} PROPERTIES OUTPUT_NAME "${target}"
        C_STANDARD 11 C_STANDARD_REQUIRED YES
        CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES
        CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    foreach(config IN ITEMS DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
    endforeach()
    if(extension STREQUAL ".cu")
        target_link_libraries(${target} PRIVATE CUDA::cudart)
        if(stem STREQUAL "perf-gemm-cuda" OR folder STREQUAL "14-cublas")
            target_link_libraries(${target} PRIVATE CUDA::cublas)
        endif()
        if(folder STREQUAL "10-dynamic-parallelism")
            # CMake/nvcc handle device-runtime linking; CUDA::cudadevrt is not an imported target.
            set_target_properties(${target} PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
        elseif(folder STREQUAL "16-profiling")
            target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
        elseif(folder STREQUAL "17-debugging")
            target_compile_options(${target} PRIVATE
                $<$<COMPILE_LANGUAGE:CUDA>:-G> $<$<COMPILE_LANGUAGE:CUDA>:-g>)
        endif()
    elseif(stem STREQUAL "perf-gemm-intelmkl")
        target_include_directories(${target} PRIVATE "${MKL_INCLUDE_DIR}")
        target_link_libraries(${target} PRIVATE
            "${MKL_intel_lp64_LIBRARY}" "${MKL_sequential_LIBRARY}" "${MKL_core_LIBRARY}"
            Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
endforeach()
